## Instalação

```bash
npx shadcn@latest add https://ds.adila.co/r/hover-card.json
```

## Uso

```tsx
import {
  HoverCard,
  HoverCardTrigger,
  HoverCardContent,
} from '@adila-sh/ui';
import { Button } from '@adila-sh/ui';

export function Example() {
  return (
    <HoverCard>
      <HoverCardTrigger render={<Button variant="link">@adila.co</Button>} />
      <HoverCardContent>
        <p className="text-sm font-medium">adila.co</p>
        <p className="text-sm text-muted-foreground">
          Design system sobre Base UI.
        </p>
      </HoverCardContent>
    </HoverCard>
  );
}
```

## Dependências
