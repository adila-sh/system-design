## Instalação

```bash
bun add @adila-sh/ui
```

## Uso

```tsx
import {
  Tooltip,
  TooltipTrigger,
  TooltipContent,
  TooltipProvider,
} from '@adila-sh/ui';
import { Button } from '@adila-sh/ui';

export function Example() {
  return (
    <TooltipProvider>
      <Tooltip>
        <TooltipTrigger render={<Button variant="outline">Passe o mouse</Button>} />
        <TooltipContent>Dica</TooltipContent>
      </Tooltip>
    </TooltipProvider>
  );
}
```
