```tsx
  SheetClose,
  SheetContent,
  SheetDescription,
  SheetFooter,
  SheetHeader,
  SheetTitle,
  SheetTrigger,
} from '@adila-sh/ui';

export function Example() {
  return (
    <Sheet>
      <SheetTrigger render={<Button variant="outline">Abrir painel</Button>} />
      <SheetContent>
        <SheetHeader>
          <SheetTitle>Editar perfil</SheetTitle>
          <SheetDescription>
            Faça alterações no seu perfil. Clique em salvar quando terminar.
          </SheetDescription>
        </SheetHeader>
        <SheetFooter>
          <Button>Salvar alterações</Button>
          <SheetClose render={<Button variant="outline">Cancelar</Button>} />
        </SheetFooter>
      </SheetContent>
    </Sheet>
  );
}
```
