## Instalação

```bash
bun add @adila-sh/ui
```

## Uso

```tsx
import {
  Field,
  FieldGroup,
  FieldLabel,
  FieldDescription,
} from '@adila-sh/ui';
import { Input } from '@adila-sh/ui';

export function Example() {
  return (
    <FieldGroup>
      <Field>
        <FieldLabel htmlFor="email">E-mail</FieldLabel>
        <Input id="email" type="email" placeholder="voce@adila.co" />
        <FieldDescription>Usaremos apenas para notificações.</FieldDescription>
      </Field>
    </FieldGroup>
  );
}
```
