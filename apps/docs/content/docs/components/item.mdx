## Instalação

```bash
bun add @adila-sh/ui
```

## Uso

```tsx
import {
  Item,
  ItemMedia,
  ItemContent,
  ItemTitle,
  ItemDescription,
  ItemActions,
} from '@adila-sh/ui';
import { Button } from '@adila-sh/ui';
import { Music } from 'lucide-react';

export function Example() {
  return (
    <Item variant="outline">
      <ItemMedia variant="icon">
        <Music />
      </ItemMedia>
      <ItemContent>
        <ItemTitle>Playlist relaxante</ItemTitle>
        <ItemDescription>32 músicas · 2h 14min</ItemDescription>
      </ItemContent>
      <ItemActions>
        <Button variant="outline" size="sm">
          Abrir
        </Button>
      </ItemActions>
    </Item>
  );
}
```

`Item` aceita `variant` (`default`, `outline`, `muted`) e `size` (`default`, `sm`, `xs`). Use `ItemGroup` com `ItemSeparator` para listas e `render` para transformar o item em link ou botão.
