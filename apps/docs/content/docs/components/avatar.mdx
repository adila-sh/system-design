## Uso

```tsx
import { Avatar, AvatarImage, AvatarFallback } from '@adila-sh/ui';

export function Example() {
  return (
    <Avatar>
      <AvatarImage src="/user.png" alt="Usuário" />
      <AvatarFallback>US</AvatarFallback>
    </Avatar>
  );
}
```
