  DrawerClose,

  return (
    <Drawer>
      <DrawerTrigger render={<Button variant="outline">Abrir Drawer</Button>} />

        <DrawerFooter>
          <Button>Confirmar</Button>
          <DrawerClose render={<Button variant="outline">Cancelar</Button>} />
        </DrawerFooter>
      </DrawerContent>
    </Drawer>
  );
}
