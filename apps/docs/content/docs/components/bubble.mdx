        <BubbleContent>Oi! Tudo bem por aí?</BubbleContent>
      </Bubble>
      <Bubble align="end">

      </Bubble>

        <BubbleContent>Também! Bora marcar aquela call?</BubbleContent>
      </Bubble>
    </BubbleGroup>
  );
}
