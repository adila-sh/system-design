## Instalação

```bash
npx shadcn@latest add https://ds.adila.co/r/dialog.json
```

## Uso

```tsx
import {
  Dialog,
  DialogContent,
  DialogHeader,
  DialogTitle,
  DialogTrigger,
} from '@adila-sh/ui';
import { Button } from '@adila-sh/ui';

export function Example() {
  return (
    <Dialog>
      <DialogTrigger render={<Button>Abrir</Button>} />
      <DialogContent>
        <DialogHeader>
          <DialogTitle>Título</DialogTitle>
        </DialogHeader>
      </DialogContent>
    </Dialog>
  );
}
```
