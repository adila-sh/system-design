
## Uso

```tsx
import { Spinner } from '@/components/ui/spinner';
import { Button } from '@/components/ui/button';

export function Example() {
  return (
    <div className="flex items-center gap-6">
      <Spinner />
      <Spinner className="size-6 text-primary" />
      <Button disabled>
        <Spinner />
        Carregando
      </Button>

}
```
