## Instalação

```bash
bun add @adila-sh/ui
```

## Uso

```tsx
import {
  Terminal,
  TerminalBody,
  TerminalLine,
  TerminalPrompt,
  TerminalCommand,
} from "@adila-sh/ui";

<Terminal>
  <TerminalBody>
    <TerminalLine>
      <TerminalPrompt>$</TerminalPrompt>
      <TerminalCommand>bun run dev</TerminalCommand>
    </TerminalLine>
  </TerminalBody>
</Terminal>;
```
