## Instalação

```bash
bun add @adila-sh/ui
```

## Uso

Defina `direction` (`horizontal` ou `vertical`) no grupo e uma altura/largura no
contêiner. `ResizableHandle` aceita `withHandle` para exibir a alça de arraste.

```tsx
import {
  ResizablePanelGroup,
  ResizablePanel,
  ResizableHandle,
} from '@adila-sh/ui';

export function Example() {
  return (
    <div className="h-48 w-full max-w-md">
      <ResizablePanelGroup direction="horizontal" className="rounded-lg border">
        <ResizablePanel defaultSize={40}>
          <div className="flex h-full items-center justify-center p-4 text-sm">Barra lateral</div>
        </ResizablePanel>
        <ResizableHandle withHandle />
        <ResizablePanel defaultSize={60}>
          <div className="flex h-full items-center justify-center p-4 text-sm">Conteúdo</div>
        </ResizablePanel>
      </ResizablePanelGroup>
    </div>
  );
}
```
