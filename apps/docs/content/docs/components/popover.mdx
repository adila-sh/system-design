## Instalação

```bash
bun add @adila-sh/ui
```

## Uso

```tsx
import { Popover, PopoverTrigger, PopoverContent } from '@adila-sh/ui';
import { Button } from '@adila-sh/ui';

export function Example() {
  return (
    <Popover>
      <PopoverTrigger render={<Button variant="outline">Abrir</Button>} />
      <PopoverContent>Conteúdo</PopoverContent>
    </Popover>
  );
}
```
