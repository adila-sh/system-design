```bash
bun add @adila-sh/ui
```

## Uso

```tsx
import {
  Combobox,
  ComboboxContent,
  ComboboxEmpty,
  ComboboxInput,
  ComboboxItem,
  ComboboxList,
} from '@adila-sh/ui';

const frameworks = ['Next.js', 'Remix', 'Astro', 'SvelteKit', 'Nuxt'];

export function Example() {
  return (
    <Combobox items={frameworks}>
      <ComboboxInput placeholder="Buscar framework..." />
      <ComboboxContent>
        <ComboboxEmpty>Nenhum resultado.</ComboboxEmpty>
        <ComboboxList>
          {(item) => (
            <ComboboxItem key={item} value={item}>
              {item}
            </ComboboxItem>
          )}
        </ComboboxList>
      </ComboboxContent>
    </Combobox>
  );
}
```
